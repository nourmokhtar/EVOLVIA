━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. 💬 Communication      → Clear, effective expression
2. 💚 Empathy            → Understanding others' feelings  
3. ⚡ Conflict Res       → Resolving disagreements
4. 🤝 Collaboration      → Working with others
5. 💪 Confidence         → Self-assurance
6. 🔄 Adaptability       → Flexibility to change

Each trait scores 0-100, starts at 50 (neutral)
Changes by -10 to +10 per user interaction

User Input
   ↓
   └─→ API Endpoint (/analyze-with-ollama)
       ↓
       └─→ Personality Service
           ↓
           ├─ Create analysis prompt
           ├─ Send to Ollama (HTTP)
           └─→ Ollama Model (Mistral)
               ↓
               └─ Analyze user text
               └─ Return trait scores (JSON)
           ↓
           ├─ Parse JSON response
           ├─ Extract trait deltas
           ├─ Update user profile
           └─ Save to database
       ↓
       └─→ Return Results to Client
           ├─ Trait changes
           ├─ Updated profile
           ├─ Analysis feedback
           └─ Success confirmation
